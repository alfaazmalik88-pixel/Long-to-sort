// Render Current Part button click handler async function renderCurrentPartWithMediaRecorder() { const video = document.getElementById('mainVideo'); const durationSelect = selectedDuration || 60; // 1 Min (60s), 1.5 Min (90s), 2 Min (120s) const startTime = (currentPart - 1) * durationSelect; const endTime = Math.min(currentPart * durationSelect, video.duration); // Screen keep-alive (Phone screen band hone se rokna) let wakeLock = null; if ('wakeLock' in navigator) { try { wakeLock = await navigator.wakeLock.request('screen'); } catch (e) {} } // 1. Processing Canvas setup const canvas = document.createElement('canvas'); canvas.width = 720; canvas.height = 1280; // 9:16 Shorts format const ctx = canvas.getContext('2d'); // 2. Audio Context (Perfect Lip-Sync) video.currentTime = startTime; await new Promise(resolve => video.onseeked = resolve); const audioCtx = new (window.AudioContext || window.webkitAudioContext)(); const sourceNode = audioCtx.createMediaElementSource(video); const audioDest = audioCtx.createMediaStreamDestination(); sourceNode.connect(audioDest); sourceNode.connect(audioCtx.destination); // 3. Combined Stream (Audio + Video) const canvasStream = canvas.captureStream(30); const combinedStream = new MediaStream([ ...canvasStream.getVideoTracks(), ...audioDest.stream.getAudioTracks() ]); const mimeType = MediaRecorder.isTypeSupported('video/mp4') ? 'video/mp4' : 'video/webm'; const recorder = new MediaRecorder(combinedStream, { mimeType: mimeType, videoBitsPerSecond: 4500000 // 1080p clean bitrate }); const recordedChunks = []; recorder.ondataavailable = (e) => { if (e.data.size > 0) recordedChunks.push(e.data); }; // UI Export Queue & Progress Bar update switchTab('export'); // Export Queue tab par redirect const progressBar = document.getElementById('renderProgressBar'); const progressText = document.getElementById('renderProgressText'); recorder.onstop = () => { if (wakeLock) wakeLock.release(); audioCtx.close(); const outputBlob = new Blob(recordedChunks, { type: mimeType }); const downloadUrl = URL.createObjectURL(outputBlob); // Original UI ka Download Button show karna showDownloadReady(downloadUrl, `Part_${currentPart}.mp4`); }; // 4. Start recording & drawing loop recorder.start(); video.muted = false; video.play(); function drawLoop() { if (video.currentTime >= endTime || video.ended) { video.pause(); recorder.stop(); return; } // Progress percentage update const currentProgress = Math.round(((video.currentTime - startTime) / (endTime - startTime)) * 100); if (progressBar) progressBar.style.width = `${currentProgress}%`; if (progressText) progressText.innerText = `${currentProgress}%`; // 9:16 Center Crop Math const vW = video.videoWidth; const vH = video.videoHeight; const targetRatio = 9 / 16; let sW = vH * targetRatio; let sH = vH; let sX = (vW - sW) / 2; let sY = 0; if (vW / vH < targetRatio) { sW = vW; sH = vW / targetRatio; sX = 0; sY = (vH - sH) / 2; } ctx.drawImage(video, sX, sY, sW, sH, 0, 0, canvas.width, canvas.height); // PART 1 Text (Bottom fixed, no dots) ctx.save(); ctx.font = "900 48px sans-serif"; ctx.fillStyle = "#ffffff"; ctx.textAlign = "center"; ctx.shadowColor = "rgba(0, 0, 0, 0.9)"; ctx.shadowBlur = 12; ctx.shadowOffsetY = 4; // Bilkul screen ke bottom par ctx.fillText(`PART ${currentPart}`, canvas.width / 2, canvas.height - 70); ctx.restore(); requestAnimationFrame(drawLoop); } drawLoop(); }
